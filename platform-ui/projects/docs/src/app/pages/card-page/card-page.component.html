Framework Usage

pui-card is a flexible surface — project any content via ng-content or use named slots (card-header, card-footer). Pass a [data] object for the automatic stat/metric layout.

Import CardComponent. Use named attribute selectors for header/footer slots, or pass [data] for the automatic metric layout.
import { CardComponent } from '@solifi/platform-ui';

@Component({
  imports: [CardComponent],
  template: `
    <!-- Slot projection -->
    <pui-card variant="default">
      <div card-header>Header</div>
      Body content here.
      <div card-footer>Footer</div>
    </pui-card>

    <!-- Stat / metric card -->
    <pui-card [data]="metric"></pui-card>

    <!-- Clickable -->
    <pui-card [clickable]="true" (cardClick)="onCardClick($event)">
      Click me — hover for lift effect.
    </pui-card>
  `
})
export class MyComponent {
  metric = { title: 'Revenue', value: '$48,200', trend: 'up', trendValue: '+12%', subtitle: 'vs last month' };
  onCardClick(e: MouseEvent) { /* ... */ }
}
Set object props (data) via a ref. Boolean and string props work as attributes.
import { useRef, useEffect } from 'react';
import '@solifi/platform-ui';

function MetricCard() {
  const ref = useRef(null);

  useEffect(() => {
    if (ref.current) {
      ref.current.data = {
        title: 'Revenue', value: '$48,200',
        trend: 'up', trendValue: '+12%', subtitle: 'vs last month'
      };
    }
  }, []);

  return <pui-card ref={ref} variant="default" />;
}

function SimpleCard() {
  return (
    <pui-card
      variant="outlined"
      size="sm"
      clickable
      onCardclick={() => console.log('clicked')}>
      Card body content
    </pui-card>
  );
}
Boolean inputs (clickable, full-width, accent) only need to be present. Set data via JS property.
<pui-card variant="outlined" id="my-card">Content</pui-card>
<pui-card variant="teal" accent clickable>Accent + clickable</pui-card>
<pui-card full-width variant="flat">Full width card</pui-card>

<script>
  document.getElementById('my-card').data = {
    title: 'Revenue', value: '$48,200', trend: 'up', trendValue: '+12%', subtitle: 'vs last month'
  };
</script>
